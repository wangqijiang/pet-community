# 交互组件使用指南

## 📦 组件导入

### 方式一：直接导入组件
```vue
<template>
  <view>
    <Picker
      v-model:visible="pickerVisible"
      title="选择宠物种类"
      :data="petTypes"
      @confirm="onPickerConfirm"
    />

    <Input
      v-model="username"
      label="用户名"
      placeholder="请输入用户名"
      :required="true"
    />

    <ActionSheet
      v-model:visible="actionSheetVisible"
      title="请选择操作"
      :actions="actions"
      @select="onActionSelect"
    />
  </view>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { Picker, Input, ActionSheet } from '@/components/interactive'

const pickerVisible = ref(false)
const actionSheetVisible = ref(false)
const username = ref('')

const petTypes = [
  [
    { label: '猫咪', value: 'cat' },
    { label: '狗狗', value: 'dog' },
    { label: '兔子', value: 'rabbit' },
    { label: '仓鼠', value: 'hamster' },
  ]
]

const actions = [
  { name: '编辑动态', type: 'default' },
  { name: '删除动态', type: 'danger' },
  { name: '分享动态', type: 'default' },
]

const onPickerConfirm = (result) => {
  console.log('选择了:', result.value, result.label)
}

const onActionSelect = (item, index) => {
  console.log('点击了:', item.name, '索引:', index)
}
</script>
```

### 方式二：使用 Composable API（推荐）
```vue
<template>
  <view>
    <button @click="showPetPicker">选择宠物种类</button>
    <button @click="showDeleteConfirm">删除动态</button>
    <button @click="showSuccessToast">显示成功提示</button>

    <Picker
      v-model:visible="picker.visible"
      :title="picker.options.title"
      :data="picker.options.data"
      @confirm="picker.options.success"
    />
  </view>
</template>

<script setup lang="ts">
import { usePicker, useDialog, useToast } from '@/composables/useComponents'

const picker = usePicker()
const dialog = useDialog()
const toast = useToast()

const petTypes = [
  [
    { label: '猫咪', value: 'cat' },
    { label: '狗狗', value: 'dog' },
    { label: '兔子', value: 'rabbit' },
  ]
]

const showPetPicker = () => {
  picker.show({
    title: '选择宠物种类',
    data: petTypes,
    success: (result) => {
      toast.success(`您选择了: ${result.label.join(', ')}`)
    }
  })
}

const showDeleteConfirm = () => {
  dialog.confirm({
    title: '确认删除',
    content: '确定要删除这条动态吗？删除后无法恢复',
    confirmText: '删除',
    confirmColor: '#FF6B8A',
    success: () => {
      toast.success('删除成功')
    }
  })
}

const showSuccessToast = () => {
  toast.success('操作成功')
}
</script>
```

## 🎯 组件使用示例

### 1. Picker 选择器

#### 单列选择
```vue
<Picker
  v-model:visible="visible"
  title="选择宠物种类"
  :data="singleColumnData"
  @confirm="onConfirm"
  @cancel="onCancel"
/>
```

```typescript
const singleColumnData = [
  [
    { label: '猫咪', value: 'cat' },
    { label: '狗狗', value: 'dog' },
    { label: '兔子', value: 'rabbit' },

  ]

const onConfirm = (result) => {
  console.log(result.value) // ['dog']
  console.log(result.label) // ['狗狗']
}
```

### 2. DatePicker 日期选择器

```vue
<DatePicker
  v-model:visible="datePickerVisible"
  title="选择宠物生日"
  :value="petBirthday"
  start-date="2000-01-01"
  end-date="2025-12-31"
  @confirm="onDateConfirm"
/>
```

```typescript
const petBirthday = ref('2023-06-15')

const onDateConfirm = (date) => {
  console.log(date) // '2023-06-15'
  toast.success(`您选择了: ${date}`)
}
```

### 3. Input 输入框

```vue
<Input
  v-model="form.name"
  label="宠物名字"
  placeholder="请输入宠物名字"
  :required="true"
  :maxlength="20"
  :clearable="true"
  prefix-icon="icon-paw"
  @focus="onFocus"
  @blur="onBlur"
/>
```

#### Props 说明
- `v-model`: 绑定的值
- `label`: 标签文字
- `placeholder`: 占位符
- `required`: 是否必填
- `maxlength`: 最大长度
- `clearable`: 是否显示清除按钮
- `prefix-icon`: 前缀图标类名
- `error`: 错误提示文字

#### Events
- `@focus`: 输入框获得焦点
- `@blur`: 输入框失去焦点
- `@confirm`: 点击完成按钮
- `@clear`: 点击清除按钮

### 4. Textarea 文本域

## 🔧 在表单中使用

```vue
<template>
  <view class="form-container">
    <Input
      v-model="form.name"
      label="宠物名字"
      placeholder="请输入宠物名字"
      :required="true"
      :error="errors.name"
    />

    <Input
      v-model="form.species"
      label="宠物种类"
      placeholder="请选择宠物种类"
      readonly
      suffix-icon="icon-arrow-right"
      @click="showSpeciesPicker"
    />

    <Input
      v-model="form.birthday"
      label="出生日期"
      placeholder="请选择出生日期"
      readonly
      suffix-icon="icon-arrow-right"
      @click="showBirthdayPicker"
    />

    <Textarea
      v-model="form.description"
      label="宠物描述"
      placeholder="介绍一下你的宠物..."
      :maxlength="200"
      :show-word-count="true"
    />

    <button class="submit-btn" @click="handleSubmit">保存</button>
  </view>
</template>

<script setup lang="ts">
import { ref, reactive } from 'vue'
import { usePicker, useDatePicker, useToast, useDialog } from '@/composables/useComponents'
import { Picker, DatePicker } from '@/components/interactive'

const picker = usePicker()
const datePicker = useDatePicker()
const toast = useToast()
const dialog = useDialog()

const form = reactive({
  name: '',
  species: '',
  birthday: '',
  description: '',
})

const errors = reactive({
  name: '',
  species: '',
})

const speciesOptions = [
  [
    { label: '猫咪', value: 'cat' },
    { label: '狗狗', value: 'dog' },
    { label: '兔子', value: 'rabbit' },
  ]
]

const showSpeciesPicker = () => {
  picker.show({
    title: '选择宠物种类',
    data: speciesOptions,
    success: (result) => {
      form.species = result.label[0]
    }
  })
}

const showBirthdayPicker = () => {
  datePicker.show({
    title: '选择出生日期',
    value: form.birthday,
    success: (result) => {
      form.birthday = result
    }
  })
}

const handleSubmit = () => {
  // 表单验证
  if (!form.name) {
    errors.name = '请输入宠物名字'
    return
  }

  if (!form.species) {
    errors.species = '请选择宠物种类'
    return
  }

  // 提交表单
  dialog.confirm({
    title: '确认提交',
    content: '确定要保存宠物信息吗？',
    success: () => {
      toast.success('保存成功')
      // 跳转到列表页
      uni.navigateBack()
    }
  })
}
</script>
```
